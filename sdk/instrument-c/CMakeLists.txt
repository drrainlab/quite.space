cmake_minimum_required(VERSION 3.16)
project(qi_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(QI_SANITIZE "Build tests with ASan+UBSan" ON)
option(QI_FUZZ "Build libFuzzer targets (clang)" OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED libsodium)

file(GLOB QI_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
add_library(qi STATIC ${QI_SRC})
target_include_directories(qi PUBLIC include ${SODIUM_INCLUDE_DIRS})
target_link_directories(qi PUBLIC ${SODIUM_LIBRARY_DIRS})
target_link_libraries(qi PUBLIC ${SODIUM_LIBRARIES})
target_compile_options(qi PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)

if(QI_SANITIZE)
  set(SAN -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(qi PRIVATE ${SAN})
  target_link_options(qi PUBLIC ${SAN})
endif()

enable_testing()
set(VEC ${CMAKE_CURRENT_SOURCE_DIR}/../../testvectors/instrument_v1.json)
set(VEC0 ${CMAKE_CURRENT_SOURCE_DIR}/../../testvectors/protocol_v0.json)
file(GLOB QI_TESTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.c)
foreach(t ${QI_TESTS})
  get_filename_component(name ${t} NAME_WE)
  add_executable(${name} ${t} tests/vectors.c)
  target_include_directories(${name} PRIVATE tests)
  target_link_libraries(${name} qi)
  target_compile_options(${name} PRIVATE -Wall -Wextra)
  if(QI_SANITIZE)
    target_compile_options(${name} PRIVATE ${SAN})
  endif()
  add_test(NAME ${name} COMMAND ${name} ${VEC} ${VEC0})
endforeach()

add_executable(qi-emit tools/qi_emit.c tests/vectors.c)
target_link_libraries(qi-emit qi)
if(QI_SANITIZE)
  target_compile_options(qi-emit PRIVATE ${SAN})
endif()

if(QI_FUZZ)
  file(GLOB QI_FUZZERS ${CMAKE_CURRENT_SOURCE_DIR}/tests/fuzz_*.c)
  foreach(f ${QI_FUZZERS})
    get_filename_component(name ${f} NAME_WE)
    add_executable(${name} ${f})
    target_link_libraries(${name} qi)
    target_compile_options(${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(${name} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
endif()
